# Everything the offline app needs at runtime.
#
# Install with:  pip install -r app/requirements.txt
#
# torch is pinned to the CPU wheel. The app never requires a GPU. If a machine
# has CUDA, torch will use it, but nothing here depends on that.
#
# Nothing in this list phones home at runtime. There is deliberately no
# requests, httpx, sentry-sdk, or analytics client anywhere in the app.

torch>=2.2,<3
torchvision>=0.17,<1
numpy>=1.24
pillow>=10.0
fastapi>=0.110
uvicorn>=0.27
python-multipart>=0.0.9

# Session B's input check (src/input_validation.py) uses these to find the
# brain-shaped region and decide whether an image is a scan at all. They are
# imported inside a function, so nothing that scans imports statically will
# notice they are missing.
#
# Leaving them out does not produce an obvious failure. The app starts, looks
# healthy, and rejects every real brain MRI with "the file could not be read as
# an image", because the validator fails closed on any exception. That exact
# bug shipped once in the PyInstaller build. Do not remove these.
scipy>=1.10
scikit-image>=0.21

# Session D's heatmap (src/explain_runtime.py) uses cv2 to resize the attention
# map on the ViT path, imported inside a function for the same reason. The
# shipped backbone is ViT, so this is the path that needs it.
opencv-python>=4.8

# DICOM, which is what every scanner in a clinic actually produces. The app
# converts it itself (app/core/dicom_loading.py) rather than asking whoever is
# at the laptop to pick a window in a viewer and export a JPEG, because that
# choice changes the picture and can change the call.
#
# Without this the app still runs: it refuses DICOM with the old instructions,
# which is a working route and a much worse one.
pydicom>=2.4

# Used by the benchmarking and validation tools in app/tools/, not by the app.
pandas>=2.0
